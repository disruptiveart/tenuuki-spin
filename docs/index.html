Tenuuki Spin

A Svelte-based 3D spinner web component for product views.

Usage

<script type="module" src="tenuuki-spin-viewer.js"></script>

<tenuuki-spin-viewer
  data-image-base-url="./frames/box-{i}.webp"
  data-image-count="60"
  data-image-number-padding="4"
  data-replacement-token="i"
  data-axis="y"
  data-direction="-1"
  data-draggable="true"
  data-initial-spin="true"
  data-spin-offset="180"
></tenuuki-spin-viewer>
